    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const searchForm = document.getElementById('searchForm');
            const searchButton = document.getElementById('searchButton');
            const alertSection = document.getElementById('alertSection');
            const resultsSection = document.getElementById('resultsSection');
            const resultsContent = document.getElementById('resultsContent');
            const resultsSummary = document.getElementById('results-summary');

            searchForm.addEventListener('submit', (e) => {
                e.preventDefault();
                handleApiSearch();
            });

            const handleApiSearch = async () => {
                setIsSearching(true);
                clearAlert();
                resultsSection.classList.add('hidden');

                const payload = {
                    sector: document.getElementById('sector').value,
                    address: document.getElementById('address').value,
                    radius_km: parseFloat(document.getElementById('radius_km').value),
                    max_results: parseInt(document.getElementById('max_results').value, 10)
                };

                try {
                    const searchResponse = await fetch('/search/start', {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload),
                    });
                    const searchResult = await searchResponse.json();
                    if (!searchResponse.ok) {
                        throw new Error(searchResult.detail || 'Ocorreu um erro ao iniciar a busca.');
                    }
                    
                    const newLeadsCount = searchResult.inseridos;

                    if (newLeadsCount > 0) {
                        showAlert(`Busca concluída! ${newLeadsCount} novos leads encontrados. A carregar resultados...`, 'success');
                        
                        const leadsResponse = await fetch(`/api/leads?limit=${newLeadsCount}`);
                        const leads = await leadsResponse.json();

                        // --- LINHA DE DEPURAÇÃO ADICIONADA ---
                        console.log("LEADS RECEBIDOS DA API:", leads);

                        displayResults(leads);
                        updateMapWithPins(leads);
                    } else {
                        showAlert('Busca concluída, mas nenhum lead novo foi encontrado. Tente refinar sua busca.', 'info');
                    }

                } catch (error) {
                    showAlert(`Erro: ${error.message}`, 'error');
                } finally {
                    setIsSearching(false);
                }
            };

            const displayResults = (leads) => {
                resultsSection.classList.remove('hidden');
                resultsSummary.textContent = `Exibindo os ${leads.length} leads encontrados nesta busca.`;
                
                if (leads.length === 0) {
                    resultsContent.innerHTML = `<p class="p-8 text-center text-gray-500">Nenhum lead encontrado.</p>`;
                    return;
                }
                
                const tableRows = leads.map(lead => `
                    <tr class="border-b border-gray-200">
                        <td class="px-6 py-4 whitespace-nowrap text-sm font-medium text-gray-900">${lead.place_name || 'N/A'}</td>
                        <td class="px-6 py-4 text-sm text-gray-600">${lead.rua || 'N/A'}, ${lead.cidade || ''}</td>
                        <!-- --- CORREÇÃO DEFENSIVA APLICADA AQUI --- -->
                        <td class="px-6 py-4 whitespace-nowrap text-sm text-gray-600">${lead.telefone || lead.phone || 'N/A'}</td>
                        <td class="px-6 py-4 whitespace-nowrap text-sm text-gray-600">${lead.total_score || 'N/A'} (${lead.reviews_count || 0})</td>
                        <td class="px-6 py-4 whitespace-nowrap text-sm font-medium">
                            <a href="${lead.url || '#'}" target="_blank" class="text-[#00AEEF] hover:underline">Ver no Mapa</a>
                        </td>
                    </tr>
                `).join('');

                const tableHTML = `
                    <div class="overflow-x-auto">
                        <table class="min-w-full divide-y divide-gray-200 results-table">
                            <thead class="bg-gray-50">
                                <tr>
                                    <th scope="col" class="px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">Nome do Local</th>
                                    <th scope="col" class="px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">Endereço</th>
                                    <th scope="col" class="px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">Telefone</th>
                                    <th scope="col" class="px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">Avaliação</th>
                                    <th scope="col" class="px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">Ações</th>
                                </tr>
                            </thead>
                            <tbody class="bg-white divide-y divide-gray-200">${tableRows}</tbody>
                        </table>
                    </div>
                `;
                resultsContent.innerHTML = tableHTML;
            };

            const setIsSearching = (isSearching) => {
                searchButton.disabled = isSearching;
                const buttonText = document.getElementById('buttonText');
                document.getElementById('searchIcon').classList.toggle('hidden', isSearching);
                document.getElementById('spinnerIcon').classList.toggle('hidden', !isSearching);
                buttonText.textContent = isSearching ? 'Buscando...' : 'Buscar';
            };
            
            const showAlert = (message, type = 'error') => {
                let bgColor = 'bg-red-100 border-red-400 text-red-700';
                if (type === 'success') bgColor = 'bg-green-100 border-green-400 text-green-700';
                if (type === 'info') bgColor = 'bg-blue-100 border-blue-400 text-blue-700';
                alertSection.innerHTML = `<div class="border ${bgColor} px-4 py-3 rounded-lg relative" role="alert"><span class="block sm:inline">${message}</span></div>`;
            };
            const clearAlert = () => { alertSection.innerHTML = ''; };
        });
    </script>

    <script>
        let map;
        let searchCircle;
        let geocoder;
        let leadMarkers = [];

        function initMap() {
            const initialPos = { lat: -10.9472, lng: -37.0731 }; 
            
            map = new google.maps.Map(document.getElementById("map"), {
                zoom: 13, center: initialPos, mapTypeControl: false, streetViewControl: false,
            });
            geocoder = new google.maps.Geocoder();
        }

        function updateMapWithPins(leads) {
            leadMarkers.forEach(marker => marker.setMap(null));
            leadMarkers = [];
            if(searchCircle) searchCircle.setMap(null);

            const bounds = new google.maps.LatLngBounds();
            const infoWindow = new google.maps.InfoWindow();

            const searchAddress = document.getElementById('address').value;
            const searchRadiusKm = parseFloat(document.getElementById('radius_km').value);
            
            geocoder.geocode({ 'address': searchAddress }, (results, status) => {
                if (status === 'OK') {
                    const searchLocation = results[0].geometry.location;
                    searchCircle = new google.maps.Circle({
                        strokeColor: "#00AEEF", strokeOpacity: 0.8, strokeWeight: 2,
                        fillColor: "#22D3EE", fillOpacity: 0.25,
                        map, center: searchLocation, radius: searchRadiusKm * 1000,
                    });
                    bounds.union(searchCircle.getBounds());
                }

                leads.forEach(lead => {
                    const leadAddress = `${lead.rua || ''}, ${lead.cidade || ''}`;
                    geocoder.geocode({ 'address': leadAddress }, (results, status) => {
                        if (status === 'OK') {
                            const location = results[0].geometry.location;
                            const leadMarker = new google.maps.Marker({
                                position: location, map: map, title: lead.place_name,
                                icon: { url: "http://maps.google.com/mapfiles/ms/icons/red-dot.png" }
                            });

                            leadMarker.addListener('click', () => {
                                // --- CORREÇÃO DEFENSIVA APLICADA AQUI ---
                                const content = `<strong>${lead.place_name}</strong><br>${leadAddress}<br>${lead.telefone || lead.phone || ''}`;
                                infoWindow.setContent(content);
                                infoWindow.open(map, leadMarker);
                            });
                            
                            leadMarkers.push(leadMarker);
                            bounds.extend(location);
                        }
                    });
                });

                setTimeout(() => {
                    if (!bounds.isEmpty()) {
                        map.fitBounds(bounds);
                    }
                }, 1500);
            });
        }
    </script>
    
    <script async defer src="https://maps.googleapis.com/maps/api/js?key=AIzaSyAyctIRjxUKbHEaruRZQxEbGK0EVH0KEe4&callback=initMap"></script>
